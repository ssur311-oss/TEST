# 깃허브 사용 첫 번째

git init (한번만 하면 됨)


여러번해도됨
git add README.md # 
git add . (모든파일)

git commit -m "first commit" (한/영 쓰고 싶은 말 다써도 됨)
여러번해도됨

git remote add origin https://github.com/ssur311-oss/test.git (한번만 하면 됨)

git push -u origin main (main : branch명)(여러번해도 됨)
git branch -m main(아무거나)(한 번만 하면 됨)
